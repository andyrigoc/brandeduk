<script>
const IMAGES=[
 "ASSET",
 "ASSET",
 "ASSET",
 "ASSET",
 "ASSET",
 "ASSET"
];
const NAMES=["LEAVERS","TOGETHER","WAREHOUSE POLOS","HI-VIS WORKWEAR","APRONS","CHEFWEAR"];
const STOPS=[
 {x:90,y:0},{x:0,y:0},{x:0,y:-90},{x:0,y:-180},{x:0,y:-270},{x:-90,y:-360}
];
const cube=document.querySelector("#cube");
const faces=[...document.querySelectorAll(".face")];
const scenes=[...document.querySelectorAll(".scene")];
const dots=document.querySelector("#dots");
const pct=document.querySelector("#pct");
const fill=document.querySelector("#fill");
const nameEl=document.querySelector("#name");

faces.forEach((f,i)=>{
 const img=new Image();
 img.src=IMAGES[i];
 img.alt="";
 f.appendChild(img);
 const b=document.createElement("button");
 b.className="dot"+(i===0?" active":"");
 b.setAttribute("aria-label",NAMES[i]);
 b.onclick=()=>scenes[i].scrollIntoView({behavior:"smooth"});
 dots.appendChild(b);
});
const dotEls=[...document.querySelectorAll(".dot")];

const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
const ease=t=>t<.5?2*t*t:-1+(4-2*t)*t;

function progress(){
 const max=document.documentElement.scrollHeight-innerHeight;
 return max>0?clamp(scrollY/max,0,1):0;
}
function activeScene(){
 const mid=innerHeight/2;
 let best=0,dist=Infinity;
 scenes.forEach((s,i)=>{
   const r=s.getBoundingClientRect();
   const d=Math.abs(r.top+r.height/2-mid);
   if(d<dist){dist=d;best=i}
 });
 return best;
}
function updateCube(p){
 const t=p*(STOPS.length-1);
 const i=Math.min(Math.floor(t),STOPS.length-2);
 const f=ease(t-i);
 const a=STOPS[i],b=STOPS[i+1];
 const x=a.x+(b.x-a.x)*f;
 const y=a.y+(b.y-a.y)*f;
 cube.style.transform=`rotateX(${x}deg) rotateY(${y}deg)`;
}
let smooth=progress(), last=-1;
function frame(){
 const target=progress();
 smooth+=(target-smooth)*.1;
 updateCube(smooth);
 const i=activeScene();
 pct.textContent=String(Math.round(smooth*100)).padStart(3,"0")+"%";
 fill.style.width=(smooth*100)+"%";
 if(i!==last){
   last=i; nameEl.textContent=NAMES[i];
   dotEls.forEach((d,n)=>d.classList.toggle("active",n===i));
 }
 requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

const io=new IntersectionObserver(entries=>{
 entries.forEach(e=>{if(e.isIntersecting)e.target.classList.add("visible")});
},{threshold:.25});
scenes.forEach(s=>io.observe(s));

document.addEventListener("click",e=>{
 const a=e.target.closest('a[href^="#"]');
 if(!a)return;
 const target=document.querySelector(a.getAttribute("href"));
 if(!target)return;
 e.preventDefault();
 target.scrollIntoView({behavior:"smooth"});
});
</script>
